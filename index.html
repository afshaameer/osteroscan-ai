    ① Model Architecture
    # bone_fracture_model.py
import torch
import torch.nn as nn
from torchvision import models

class FractureClassifier(nn.Module):
    def __init__(self, num_bone_classes=6):
        super().__init__()
        # Load pretrained ResNet-50 backbone
        backbone = models.resnet50(pretrained=True)
        self.features = nn.Sequential(*list(backbone.children())[:-1])

        # Shared FC layer
        self.shared = nn.Sequential(
            nn.Flatten(),
            nn.Linear(2048, 512),
            nn.ReLU(),
            nn.Dropout(0.4)
        )
        # Head 1: Fracture detection (binary)
        self.fracture_head = nn.Linear(512, 1)
        # Head 2: Bone type classification
        self.bone_head = nn.Linear(512, num_bone_classes)
        # Head 3: Severity estimation
        self.severity_head = nn.Linear(512, 3)

    def forward(self, x):
        x = self.features(x)
        x = self.shared(x)
        return {
            'fracture': self.fracture_head(x).squeeze(1),
            'bone':     self.bone_head(x),
            'severity': self.severity_head(x)
        }

    ③ Training Loop
    def train_epoch(model, loader, optimizer, device):
    model.train()
    bce = nn.BCEWithLogitsLoss()
    ce  = nn.CrossEntropyLoss()

    for imgs, labels in loader:
        imgs = imgs.to(device)
        optimizer.zero_grad()
        out = model(imgs)

        # Multi-task loss (weighted sum)
        loss = (
            1.0 * bce(out['fracture'], labels['fracture'].float()) +
            0.6 * ce(out['bone'], labels['bone']) +
            0.4 * ce(out['severity'], labels['severity'])
        )
        loss.backward()
        optimizer.step()

# Setup: AdamW + cosine LR schedule
optimizer = torch.optim.AdamW(model.parameters(), lr=1e-4, weight_decay=1e-4)
scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=30)
    ④ Evaluation Metrics
    from sklearn.metrics import (
    accuracy_score, confusion_matrix,
    classification_report, roc_auc_score
)

def evaluate(model, loader, device):
    model.eval()
    all_preds, all_labels = [], []
    with torch.no_grad():
        for imgs, labels in loader:
            out = model(imgs.to(device))
            preds = (torch.sigmoid(out['fracture']) > 0.5).cpu()
            all_preds.extend(preds.tolist())
            all_labels.extend(labels['fracture'].tolist())

    print(classification_report(all_labels, all_preds,
          target_names=['Normal', 'Fractured']))
    print("Confusion Matrix:\n",
          confusion_matrix(all_labels, all_preds))
    print("AUC-ROC:", roc_auc_score(all_labels, all_preds))

    Model Improvement Strategies
    DATA STRATEGY
    • Use MURA or FracAtlas datasets (1000+ X-rays)
    • Apply CLAHE contrast enhancement for X-rays
    • Augment with elastic distortions & Gaussian noise
    • Balance classes with oversampling or focal loss
    ARCHITECTURE
    • Try EfficientNet-B4 for better accuracy/size tradeoff
    • Add Grad-CAM for visual explainability
    • Use attention gates to focus on fracture region
    • Ensemble ResNet + DenseNet predictions
    TRAINING
    • Freeze backbone layers initially, unfreeze gradually
    • Use label smoothing (0.1) to reduce overconfidence
    • Mixed precision training (fp16) for faster training
    • Test-time augmentation (TTA) for better inference
    EVALUATION
    • Track F1-score, not just accuracy (imbalanced data)
    • Use 5-fold cross-validation for robust metrics
    • Report sensitivity & specificity separately
    • Calibrate model confidence with temperature scaling
